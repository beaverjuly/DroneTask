  // Warm browser image decode cache. jsPsych preloading fetches assets,
  // but some browsers still defer decode until first render, which can
  // cause a brief flicker between instruction pages.
  async function warmImageCache(imagePaths) {
    const uniquePaths = Array.from(new Set(imagePaths || []));
    await Promise.all(uniquePaths.map((src) => {
      return new Promise((resolve) => {
        const img = new Image();
        img.onload = function() {
          if (typeof img.decode === 'function') {
            img.decode().catch(() => {}).finally(resolve);
          } else {
            resolve();
          }
        };
        img.onerror = function() { resolve(); };
        img.src = src;
      });
    }));
  }

  // Keep frequently reused instruction images mounted offscreen so the
  // browser does not repeatedly evict/re-rasterize them between pages.
  // This targets the brief practice-page flicker seen on some clients.
  function pinImagesOffscreen(imagePaths) {
    const uniquePaths = Array.from(new Set(imagePaths || []));
    const holder = document.createElement('div');
    holder.id = 'img-pin-cache';
    holder.setAttribute('aria-hidden', 'true');
    holder.style.position = 'fixed';
    holder.style.left = '-10000px';
    holder.style.top = '-10000px';
    holder.style.width = '1px';
    holder.style.height = '1px';
    holder.style.overflow = 'hidden';
    holder.style.opacity = '0';
    holder.style.pointerEvents = 'none';

    uniquePaths.forEach((src) => {
      const img = new Image();
      img.src = src;
      img.loading = 'eager';
      img.decoding = 'sync';
      // Tiny rendered size is enough to keep it in the render pipeline
      // without affecting layout.
      img.width = 2;
      img.height = 2;
      holder.appendChild(img);
    });

    document.body.appendChild(holder);
    window.__imgPinCache = holder;
  }

  // Rewrite instruction image paths to in-memory blob URLs. This avoids
  // revalidation/network jitter for JPGs that appear across many
  // instruction pages and can remove residual flicker on some clients.
  async function localizeInstructionImageSources(imagePaths) {
    const uniquePaths = Array.from(new Set(imagePaths || []));
    const urlMap = {};

    await Promise.all(uniquePaths.map(async (src) => {
      try {
        const res = await fetch(src, { cache: 'force-cache' });
        const blob = await res.blob();
        urlMap[src] = URL.createObjectURL(blob);
      } catch (e) {
        // Keep original path on failure.
        urlMap[src] = src;
      }
    }));

      inst_summary, inst1, inst2, inst3, inst4, inst5, inst7,

      if (!node || !Array.isArray(node.pages)) return;
      node.pages = node.pages.map((html) => {
        let next = html;
        uniquePaths.forEach((src) => {
          const localized = urlMap[src] || src;
          next = next.split(src).join(localized);
        });
        return next;
      });
    });

    // Keep refs available so they can be revoked later if needed.
    window.__instructionBlobUrls = Object.values(urlMap).filter((u) => u && u.startsWith('blob:'));
  }

  // Wrap instructions trials so transitions hide until image nodes are ready.
  function addSmoothTransitions(node) {
    if (!node || node.type !== 'instructions') return node;

    const originalOnStart = node.on_start;
    const originalOnLoad = node.on_load;

    node.on_start = function(trial) {
      const display = document.querySelector('.jspsych-display-element');
      if (display) display.classList.add('transitioning');
      if (typeof originalOnStart === 'function') {
        originalOnStart.call(this, trial);
      }
    };

    node.on_load = function() {
      const display = document.querySelector('.jspsych-display-element');
      if (!display) {
        if (typeof originalOnLoad === 'function') originalOnLoad.call(this);
        return;
      }

      const imgs = Array.from(display.querySelectorAll('img'));
      Promise.all(imgs.map((img) => {
        if (img.complete && img.naturalWidth > 0) {
          return (typeof img.decode === 'function') ? img.decode().catch(() => {}) : Promise.resolve();
        }
        return new Promise((resolve) => {
          img.addEventListener('load', () => {
            if (typeof img.decode === 'function') {
              img.decode().catch(() => {}).finally(resolve);
            } else {
              resolve();
            }
          }, { once: true });
          img.addEventListener('error', resolve, { once: true });
        });
      })).finally(() => {
        display.classList.remove('transitioning');
      });

      if (typeof originalOnLoad === 'function') {
        originalOnLoad.call(this);
      }
    };

    return node;
  }

  // sequence after each block, then append the memory test for that
  // block. Once the combined trials array is constructed, we
  // assemble the overall timeline and initialise jsPsych. This
  // ensures that the discounting task runs immediately after each
  // bird block and before the memory task, without altering the
  // existing instruction and practice logic.
  (async function() {
    // Compute the number of experimental blocks (baseline + 4
    // volatility/stochasticity conditions). This should match the
    // length of the randomized array defined earlier.
    var num_blocks = randomized.length;
    if (num_blocks > 0) {
      var block_length = Math.floor(trials.length / num_blocks);
      var new_trials = [];
      // The baseline delay‑discounting phase has been removed in MOD12.
      // Participants will encounter the delay discounting task only after
      // completing each bird block.
      for (var b = 0; b < num_blocks; b++) {
        var start_idx = b * block_length;
        var end_idx = start_idx + block_length;
        var block_trials = trials.slice(start_idx, end_idx);
        // Append the original bird trials for this block
        new_trials = new_trials.concat(block_trials);
        // Derive a descriptive condition tag using the true volatility
        // and stochasticity factors if available. This tag is
        // recorded with each discounting trial for analysis.
        var conditionTag;
        try {
          if (typeof factors_vol !== 'undefined' && typeof factors_stc !== 'undefined') {
            conditionTag = 'vol' + factors_vol[randomized[b]] + '_stc' + factors_stc[randomized[b]];
          } else {
            conditionTag = randomized[b];
          }
        } catch (err) {
          conditionTag = randomized[b];
        }
        // Append the delay discounting timeline for this block. Await
        // the Promise returned by runDelayDiscounting so that the
        // discounting trials are fully defined before continuing.
        // Set the global land background for the upcoming discounting trials to match
        // the current bird block.  The discounting plugin reads window.current_land_img
        // to display this background.
        window.current_land_img = block_land_backgrounds[randomized[b] % block_land_backgrounds.length];
        var discount_tl = await runDelayDiscounting(conditionTag);
        new_trials = new_trials.concat(discount_tl);
        // Append the memory test timeline for this block. The memory
        // pairs are shuffled anew for each block within
        // create_memory_timeline().
        var mem_tl = create_memory_timeline(b + 1);
        new_trials = new_trials.concat(mem_tl);
      }
      trials = new_trials;
    }
    // Assemble the main experiment timeline. We preserve the order
    // established in the original implementation: welcome page,
    // feedback0, instruction/practice sequences, the combined trials,
    // final feedback, and finish. Note that feedback0 and
    // Demographics survey (shown after all 4 blocks complete)
    var survey_demographics = {
      type: 'survey-demo',
      button_label: 'Continue'
    };

    // instructions_loop are defined earlier in the script.
    var timeline = [];
    timeline = timeline.concat(welcome);
    timeline = timeline.concat(feedback0);
    timeline = timeline.concat(instructions_loop);
    timeline = timeline.concat(trials);
    timeline = timeline.concat(survey_demographics);
    timeline = timeline.concat(post_game_notes);
    timeline = timeline.concat(feedback1);
    timeline = timeline.concat(finish);
    // Initialise jsPsych with the constructed timeline. Use the same
    // settings for the progress bar and preloading as in the MOD9
    // version. The on_finish handler also mirrors the original
    // functionality, saving the data and redirecting participants
    // based on data quality.
    // Ensure instruction/game images are decoded before timeline starts.
    await warmImageCache(preload_images);

    // Specifically pin instruction/practice JPG assets that are shown in
    // rapid succession where participants reported visible flicker.
    var instructionJpgs = [
      '../static/img/imga1.jpg',
      '../static/img/imga2.jpg',
      '../static/img/imgb1.jpg',
      '../static/img/imgb2.jpg',
      '../static/img/imgb3.jpg',
      '../static/img/imgc1.jpg',
      '../static/img/imgc2.jpg',
      '../static/img/imgc3.jpg',
      '../static/img/imgd1.jpg',
      '../static/img/imgd2.jpg',
      '../static/img/imgd3.jpg',
      '../static/img/imge1.jpg'
    ];

    pinImagesOffscreen(instructionJpgs);
    await localizeInstructionImageSources(instructionJpgs);

    jsPsych.init({
      timeline: timeline,
      show_progress_bar: true,
      preload_images: preload_images, // , preload_rpm_images
      on_finish: function() {

        // Remove requirement to verify redirect
        window.removeEventListener("beforeunload", verify_unload);

        // Add interactions to the data variable
        var interaction_data = jsPsych.data.getInteractionData();
        jsPsych.data.get().addToLast({ interactions: interaction_data.json() });

        if (low_quality) {
          // Save rejected dataset to disk + redirect w/ decoy code (Prolific)
          redirect_reject("{{workerId}}", "{{assignmentId}}", "{{hitId}}", "{{code_reject}}");
          // (Alternative MTurk/error-style: redirect_reject("1011");)
        } else {
          // Save complete dataset to disk + redirect w/ real completion code
          redirect_success("{{workerId}}", "{{assignmentId}}", "{{hitId}}", "{{code_success}}");
        }
      }
    });
  })();

</script>
